# T1 拉格朗日中值定理：平行弦与切线
f(x) = sin(x)
inputF1 = InputBox(f)
aM = Slider(0, 2, 0.05)
SetValue(aM, 0)
bM = Slider(0.5, 3.14, 0.05)
SetValue(bM, 3.14)
A1 = (aM, f(aM))
B1 = (bM, f(bM))
secM = Line(A1, B1)
sM = Slope(secM)
cM = Slider(0, 3.14, 0.02)
SetValue(cM, 1.57)
C1 = (cM, f(cM))
tanM = Tangent(C1, f)
sTan = Slope(tanM)
dfM = Derivative(f)
SetVisibleInView(dfM, 1, false)
kM(x) = dfM(x) - sM
mvtX = Root(kM, aM, bM)
mvtM = (mvtX, f(mvtX))
txtT1 = Text("割线斜率 = " + sM + " \\ 切线斜率 = " + sTan, (-0.6, 0.7), true, true)
SetColor(f, 0.2, 0.4, 0.9)
SetColor(secM, 0.9, 0.3, 0.1)
SetColor(tanM, 0.1, 0.6, 0.3)
SetColor(mvtM, 0.6, 0.1, 0.8)
SetColor(A1, 0.9, 0.3, 0.1)
SetColor(B1, 0.9, 0.3, 0.1)
ShowLabel(mvtM, false)
ShowLabel(sM, false)
ShowLabel(sTan, false)
btnMvt = Button("找到中值点")
SetCoords(btnMvt, 430, 70)
