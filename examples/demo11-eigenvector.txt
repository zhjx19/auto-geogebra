# T4 特征向量：变换后方向不变的向量
a11 = Slider(-2, 2, 0.1)
SetValue(a11, 2)
a12 = Slider(-2, 2, 0.1)
SetValue(a12, 1)
a21 = Slider(-2, 2, 0.1)
SetValue(a21, 1)
a22 = Slider(-2, 2, 0.1)
SetValue(a22, 2)
M4 = {{a11, a12}, {a21, a22}}
tr4 = a11 + a22
det4 = a11 * a22 - a12 * a21
disc4 = tr4^2 - 4 * det4
root4 = If(disc4 >= 0, sqrt(disc4), 0)
lam1 = (tr4 + root4) / 2
lam2 = (tr4 - root4) / 2
e1 = If(a12 != 0, (a12, lam1 - a11), (lam1 - a22, a21))
e2 = If(a12 != 0, (a12, lam2 - a11), (lam2 - a22, a21))
l1 = Line((0, 0), e1)
l2 = Line((0, 0), e2)
th4 = Slider(0, 360, 1)
SetValue(th4, 30)
P4 = (cos(th4°), sin(th4°))
Pimg4 = ApplyMatrix(M4, P4)
ray4 = Line((0, 0), P4)
rayImg4 = Line((0, 0), Pimg4)
cr4 = x(P4) * y(Pimg4) - y(P4) * x(Pimg4)
txtT4 = Text("方向角 = " + th4 + "° \\ 叉积 = " + cr4 + "（≈0 即特征方向）\\ 特征值 = " + lam1 + ", " + lam2, (-2.7, 2.5), true, true)
SetColor(ray4, 0.5, 0.5, 0.5)
SetColor(rayImg4, 0.15, 0.45, 0.9)
SetColor(l1, 0.1, 0.7, 0.2)
SetColor(l2, 0.1, 0.7, 0.2)
ShowLabel(cr4, false)
ShowLabel(l1, false)
ShowLabel(l2, false)
ShowLabel(tr4, false)
ShowLabel(det4, false)
ShowLabel(disc4, false)
ShowLabel(root4, false)
ShowLabel(e1, false)
ShowLabel(e2, false)
