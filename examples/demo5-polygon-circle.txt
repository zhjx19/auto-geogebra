# 正 n 边形逼近圆——交互升级版（滑块 + 播放/停止/重置按钮）
n = Slider(3, 60, 1)
SetValue(n, 6)
circleUnit = Circle((0, 0), 1)
polyN = Polygon(Sequence((cos(2 * pi * k / n), sin(2 * pi * k / n)), k, 1, n))
areaN = Area(polyN)
txtArea = Text("n = " + n + "，多边形面积 = " + areaN + " → π ≈ 3.1416", (-2.9, 2.6), true, true)
SetColor(circleUnit, 0.6, 0.6, 0.6)
SetColor(polyN, 0.2, 0.4, 1)
ShowLabel(polyN, false)
ShowLabel(areaN, false)
btnAnimN = Button("播放动画")
btnStopN = Button("停止")
btnResetN = Button("重置")
SetCoords(btnAnimN, 420, 70)
SetCoords(btnStopN, 420, 120)
SetCoords(btnResetN, 420, 170)
