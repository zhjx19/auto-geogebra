# T5 置信区间的覆盖率模拟
mu0 = Slider(-2, 2, 0.1)
SetValue(mu0, 0)
sg5 = 1
nn5 = Slider(5, 50, 1)
SetValue(nn5, 20)
mrep5 = Slider(10, 60, 1)
SetValue(mrep5, 40)
tick5 = 0
zc5 = 1.96
h5 = 10 / mrep5
samples5 = Sequence(Sequence(RandomNormal(mu0, sg5) + 0 * tick5, j, 1, nn5), i, 1, mrep5)
means5 = Sequence(Mean(Element(samples5, i)), i, 1, mrep5)
moe5 = zc5 * sg5 / sqrt(nn5)
segs5 = Zip(Segment((mm - moe5, k * h5), (mm + moe5, k * h5)), mm, means5, k, Sequence(1, mrep5))
flags5 = Zip(If(abs(mm - mu0) <= moe5, 1, 0), mm, means5)
covN5 = Sum(flags5)
covRate5 = covN5 / mrep5
vline5 = Segment((mu0, 0), (mu0, 10))
txtT5 = Text("样本量 n = " + nn5 + " \\ 置信区间覆盖率 = " + covRate5 + "（理论 0.95）", (-2.9, 9.7), true, true)
SetColor(vline5, 0.9, 0.2, 0.1)
SetColor(segs5, 0.2, 0.4, 0.9)
SetColor(moe5, 0, 0, 0)
ShowLabel(moe5, false)
ShowLabel(covN5, false)
ShowLabel(covRate5, false)
ShowLabel(flags5, false)
ShowLabel(tick5, false)
ShowLabel(h5, false)
ShowLabel(means5, false)
ShowLabel(samples5, false)
ZoomIn(-3, -0.5, 3, 10.5)
btnRedraw5 = Button("重新抽样")
SetCoords(btnRedraw5, 430, 70)
