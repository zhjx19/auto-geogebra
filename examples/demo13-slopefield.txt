# T6 方向场与积分曲线

P0 = (0, 0.3)

txtT6 = Text("y' = y - y² \\ 拖动起点 P0 观察积分曲线", (-4.5, 4.2), true, true)

SetColor(sol6, 0.1, 0.4, 0.9)

ShowLabel(sf6, false)
