# T3 行列式作为面积缩放因子
a11 = Slider(-2, 2, 0.1)
SetValue(a11, 1.5)
a12 = Slider(-2, 2, 0.1)
SetValue(a12, 0.5)
a21 = Slider(-2, 2, 0.1)
SetValue(a21, 0.3)
a22 = Slider(-2, 2, 0.1)
SetValue(a22, 1.4)
M3 = {{a11, a12}, {a21, a22}}
sq3 = Polygon((0, 0), (1, 0), (1, 1), (0, 1))
sqImg3 = ApplyMatrix(M3, sq3)
det3 = Determinant(M3)
areaImg3 = Area(sqImg3)
txtT3 = Text("行列式 = " + det3 + " \\ 像的面积 = " + areaImg3, (-2.6, 2.4), true, true)
SetColor(sq3, 0.5, 0.5, 0.5)
SetColor(sqImg3, 0.15, 0.45, 0.9)
ShowLabel(sqImg3, false)
ShowLabel(areaImg3, false)
ShowLabel(det3, false)
