# T2 方向导数与梯度方向的关系（3D）
f2(x, y) = x^2 + 3 * y^2
x0 = Slider(-2, 2, 0.1)
SetValue(x0, 1)
y0 = Slider(-2, 2, 0.1)
SetValue(y0, 0.5)
th2 = Slider(0, 360, 5)
SetValue(th2, 0)
P3 = (x0, y0, f2(x0, y0))
wx2 = Derivative(f2, x)
wy2 = Derivative(f2, y)
SetVisibleInView(wx2, -1, false)
SetVisibleInView(wy2, -1, false)
SetVisibleInView(wx2, 1, false)
SetVisibleInView(wy2, 1, false)
gx2 = wx2(x0, y0)
gy2 = wy2(x0, y0)
du2 = gx2 * cos(th2°) + gy2 * sin(th2°)
gmod2 = sqrt(gx2^2 + gy2^2)
gvec2 = Vector((x0, y0, f2(x0, y0)), (x0 + gx2, y0 + gy2, f2(x0, y0)))
uvec2 = Vector((x0, y0, f2(x0, y0)), (x0 + cos(th2°), y0 + sin(th2°), f2(x0, y0)))
txtT2 = Text("方向导数 = " + du2 + " \\ 梯度模 = " + gmod2, (-2, -2, 6), true, true)
SetColor(f2, 0.3, 0.5, 0.9)
SetColor(gvec2, 0.9, 0.2, 0.1)
SetColor(uvec2, 0.1, 0.7, 0.2)
ShowLabel(gvec2, false)
ShowLabel(uvec2, false)
ShowLabel(gx2, false)
ShowLabel(gy2, false)
ShowLabel(du2, false)
ShowLabel(gmod2, false)
btnDir2 = Button("对准梯度方向")
SetCoords(btnDir2, 430, 70)
